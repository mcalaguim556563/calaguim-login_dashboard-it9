app/Http/Controllers/AppointmentController.php
<?php
namespace App\Http\Controllers;
use App\Models\Appointment;
use Illuminate\Http\Request;
class AppointmentController extends Controller
{
    public function index()
    {
        $appointments = Appointment::latest()->paginate(10);
        return view('appointments.index', compact('appointments'));
    }
    public function create()
    {
        return view('appointments.create');
    }
    public function store(Request $request)
    {
        $request->validate([
            'patient_name' => 'required|string|max:255',
            'doctor_name' => 'required|string|max:255',
            'session_count' => 'required|integer|min:1',
            'status' => 'required|in:scheduled,completed,cancelled',
            'appointment_date' => 'required|date',
            'notes' => 'nullable|string',
        ]);
        Appointment::create($request->all());
        return redirect()->route('appointments.index')->with('success', 'Record added!');
    }
    public function edit(Appointment $appointment)
    {
        return view('appointments.edit', compact('appointment'));
    }
    public function update(Request $request, Appointment $appointment)
    {
        $request->validate([
            'patient_name' => 'required|string|max:255',
            'doctor_name' => 'required|string|max:255',
            'session_count' => 'required|integer|min:1',
            'status' => 'required|in:scheduled,completed,cancelled',
            'appointment_date' => 'required|date',
            'notes' => 'nullable|string',
        ]);
        $appointment->update($request->all());
        return redirect()->route('appointments.index')->with('success', 'Record updated!');
    }
    public function destroy(Appointment $appointment)
    {
        $appointment->delete();
        return redirect()->route('appointments.index')->with('success', 'Record deleted!');
    }
}

app/Http/Controllers/AppointmentReportController.php
<?php
namespace App\Http\Controllers;
use App\Models\Appointment;
use Barryvdh\DomPDF\Facade\Pdf;
class AppointmentReportController extends Controller
{
    private function getStats()
    {
        $doctors = Appointment::select('doctor_name')->distinct()->pluck('doctor_name');
        $summary = [];
        foreach ($doctors as $doctor) {
            $summary[$doctor] = [
                'total' => Appointment::where('doctor_name', $doctor)->sum('session_count'),
                'scheduled' => Appointment::where('doctor_name', $doctor)->where('status', 'scheduled')->sum('session_count'),
                'completed' => Appointment::where('doctor_name', $doctor)->where('status', 'completed')->sum('session_count'),
                'cancelled' => Appointment::where('doctor_name', $doctor)->where('status', 'cancelled')->sum('session_count'),
            ];
        }
        return [
            'totalRecords' => Appointment::count(),
            'totalScheduled' => Appointment::where('status', 'scheduled')->sum('session_count'),
            'totalCompleted' => Appointment::where('status', 'completed')->sum('session_count'),
            'totalCancelled' => Appointment::where('status', 'cancelled')->sum('session_count'),
            'doctorSummary' => $summary,
            'appointments' => Appointment::latest()->get(),
        ];
    }
    public function index()
    {
        return view('appointment-reports.index', $this->getStats());
    }
    public function exportPdf()
    {
        $data = $this->getStats();
        $data['dateGenerated'] = now()->format('F d, Y h:i A');
        $pdf = Pdf::loadView('appointment-reports.pdf', $data)->setPaper('a4', 'landscape');
        return $pdf->download('appointment-report-' . now()->format('Y-m-d') . '.pdf');
    }
    public function exportExcel()
    {
        $appointments = Appointment::latest()->get();
        $spreadsheet = new \PhpOffice\PhpSpreadsheet\Spreadsheet();
        $sheet = $spreadsheet->getActiveSheet();
        $sheet->setCellValue('A1', 'Clinic Appointment Report - ' . now()->format('F d, Y'));
        $sheet->mergeCells('A1:G1');
        $headers = ['#', 'Patient Name', 'Doctor Name', 'Sessions', 'Status', 'Appointment Date', 'Notes'];
        foreach ($headers as $col => $h) {
            $sheet->setCellValueByColumnAndRow($col + 1, 3, $h);
        }
        foreach ($appointments as $i => $a) {
            $row = $i + 4;
            $sheet->setCellValueByColumnAndRow(1, $row, $i + 1);
            $sheet->setCellValueByColumnAndRow(2, $row, $a->patient_name);
            $sheet->setCellValueByColumnAndRow(3, $row, $a->doctor_name);
            $sheet->setCellValueByColumnAndRow(4, $row, $a->session_count);
            $sheet->setCellValueByColumnAndRow(5, $row, ucfirst($a->status));
            $sheet->setCellValueByColumnAndRow(6, $row, $a->appointment_date->format('Y-m-d'));
            $sheet->setCellValueByColumnAndRow(7, $row, $a->notes ?? '');
        }
        $writer = new \PhpOffice\PhpSpreadsheet\Writer\Xlsx($spreadsheet);
        return response()->streamDownload(function () use ($writer) {
            $writer->save('php://output');
        }, 'appointment-report-' . now()->format('Y-m-d') . '.xlsx', [
            'Content-Type' => 'application/vnd.openxmlformatsofficedocument.spreadsheetml.sheet',
        ]);
    }
}
